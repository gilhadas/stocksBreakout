# Breakout Scanner Cron Jobs
# Add these to your crontab with: crontab -e
#
# IMPORTANT: All times are US Eastern (NYSE/NASDAQ market hours)
# Add "TZ=America/New_York" line at the top of your crontab to ensure
# correct timing regardless of server timezone.
# Format: minute hour day month weekday command
#
# HEALTHCHECKS.IO SETUP (free remote monitoring):
# 1. Sign up at https://healthchecks.io (free: 20 checks)
# 2. Create checks: longterm, swing, daytrade, monitor, validate
# 3. Replace HC_UUID_xxx placeholders below with your real UUIDs
# 4. Install mobile app for push notifications on failures
# Ping format: curl -fsS -m 10 --retry 3 https://hc-ping.com/UUID

TZ=America/New_York
HC_BASE=https://hc-ping.com

PROJECT_ROOT=$( [ -d "$HOME/documents/github/stocksbreakout" ] && echo "$HOME/documents/github/stocksbreakout" || echo "$HOME/stocksbreakout" )
PYTHON_BIN=$PROJECT_ROOT/venv/bin/python3

# ============================================
# LONG-TERM POSITION TRADING (Weekly scans)
# ============================================

# Every Monday at 9:00 AM - Phase 1: S&P 500 wide scan, export PREMIUM tickers
0 9 * * 1 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py "input/ALL.txt" --mode longterm --bounce --sector-buzz --sentiment --export-premium input/premium_longterm.txt --live --cron --notify >> scanner_output/logs/cron_longterm.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_LONGTERM > /dev/null

# Every Monday at 9:15 AM - Long-term exit evaluation
15 9 * * 1 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py input/1_26_setups.txt --mode longterm --exit-file input/positions_longterm.csv --live --cron --notify >> scanner_output/logs/cron_longterm_exit.log 2>&1


# ============================================
# SWING TRADING (Daily scans)
# ============================================

# Every weekday at 9:35 AM (after market open) - Phase 1: S&P 500 wide scan, export PREMIUM tickers
35 9 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py "input/ALL.txt" --mode swing --bounce --sector-buzz --sentiment --auto-positions input/positions_swing_mock.csv --export-premium input/premium_swing.txt --live --cron --notify >> scanner_output/logs/cron_swing.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_SWING > /dev/null

# Every weekday at 3:45 PM (before market close) - Swing exit evaluation (mock positions)
45 15 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py input/1_26_setups.txt --mode swing --exit-file input/positions_swing_mock.csv --live --cron --notify >> scanner_output/logs/cron_swing_exit.log 2>&1

# Every weekday at 4:30 PM (after market close) - Phase 2: Re-evaluate PREMIUM + exit check
30 16 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py input/premium_swing.txt --mode swing --bounce --exit-file input/positions_swing_mock.csv --both --sentiment --auto-positions input/positions_swing_mock.csv --live --cron --notify >> scanner_output/logs/cron_swing_combined.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_SWING > /dev/null

# Every weekday at 7:30 PM - Phase 2: Evening PREMIUM re-evaluation
30 19 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py input/premium_swing.txt --mode swing --bounce --sentiment --auto-positions input/positions_swing_mock.csv --live --cron --notify >> scanner_output/logs/cron_swing_evening.log 2>&1


# ============================================
# DAY TRADING (Intraday scans)
# ============================================

# Every weekday at 9:35 AM - Phase 1: S&P 500 wide scan, export PREMIUM + momentum-watch tickers
# momentum_watch_daytrade.txt = PREMIUM/GOLD + HIGH-momentum + near-miss high-vol (for Phase 2 re-scans)
35 9 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py "input/ALL.txt" --mode daytrade --bounce --sector-buzz --sentiment --auto-positions input/positions_daytrade_mock.csv --export-premium input/premium_daytrade.txt --export-momentum-watch input/momentum_watch_daytrade.txt --live --cron --notify >> scanner_output/logs/cron_daytrade_morning.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_DAYTRADE > /dev/null

# Every weekday at 10:00 AM - Phase 2: Re-evaluate momentum-watch tickers (PREMIUM + HIGH-mom + near-miss)
0 10 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py input/momentum_watch_daytrade.txt --mode daytrade --bounce --sentiment --auto-positions input/positions_daytrade_mock.csv --live --cron --notify >> scanner_output/logs/cron_daytrade_mid.log 2>&1

# Every weekday at 2:00 PM - Phase 2: Re-evaluate momentum-watch tickers
0 14 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py input/momentum_watch_daytrade.txt --mode daytrade --bounce --sentiment --auto-positions input/positions_daytrade_mock.csv --live --cron --notify >> scanner_output/logs/cron_daytrade_afternoon.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_DAYTRADE > /dev/null

# Every weekday at 3:30 PM - Day trade exit check (mock positions)
30 15 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py input/1_26_setups.txt --mode daytrade --exit-file input/positions_daytrade_mock.csv --live --cron --notify >> scanner_output/logs/cron_daytrade_exit.log 2>&1

# Every weekday at 8:20 PM - Phase 4: Evening re-evaluate momentum-watch tickers
20 20 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py input/momentum_watch_daytrade.txt --mode daytrade --bounce --sentiment --auto-positions input/positions_daytrade_mock.csv --live --cron --notify >> scanner_output/logs/cron_daytrade_evening.log 2>&1

# ============================================
# MOMENTUM-WATCH MONITOR (Every 15 minutes)
# ============================================
# Validates signaled stocks: HOLDING / FADING / FAILED
# Reads input/momentum_watch_daytrade.txt (exported by Phase 1 daytrade scan)
# Sends Discord alert on first status change to FADING or FAILED

# Every weekday at 9:45 AM - First check after Phase 1 scan (watch list just populated)
45 9 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN monitor_watch.py --notify >> scanner_output/logs/cron_watch_monitor.log 2>&1

# Every 15 minutes from 10:00 AM to 3:45 PM - Continuous momentum validation
*/15 10-15 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN monitor_watch.py --notify >> scanner_output/logs/cron_watch_monitor.log 2>&1

# Every weekday at 4:00 PM - Final check at market close
0 16 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN monitor_watch.py --notify >> scanner_output/logs/cron_watch_monitor.log 2>&1


# ============================================
# PORTFOLIO MONITORING (Every 15 minutes)
# ============================================
# Lightweight price checks on open positions — alerts on drops toward stop loss
# Monitors both swing and daytrade positions

# Every weekday at 9:30 AM - Reset alert history for new trading day
30 9 * * 1-5 cd /users/gilhadas/documents/github/stocksbreakout && mv scanner_output/.monitor_alerts.txt scanner_output/.monitor_alerts_$(date +\%Y\%m\%d).txt 2>/dev/null || true

# Every weekday at 9:45 AM - First check after market open
45 9 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py "input/ALL.txt" --monitor input/positions_swing_mock.csv,input/positions_daytrade_mock.csv --live --cron --notify >> scanner_output/logs/cron_monitor.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_MONITOR > /dev/null

# Every 15 minutes from 10:00 AM to 3:45 PM - Continuous monitoring
*/15 10-15 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py "input/ALL.txt" --monitor input/positions_swing_mock.csv,input/positions_daytrade_mock.csv --live --cron --notify >> scanner_output/logs/cron_monitor.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_MONITOR > /dev/null

# Every weekday at 4:00 PM - Final check at market close
0 16 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN breakout_scanner.py "input/ALL.txt" --monitor input/positions_swing_mock.csv,input/positions_daytrade_mock.csv --live --cron --notify >> scanner_output/logs/cron_monitor.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_MONITOR > /dev/null


# ============================================
# SIGNAL VALIDATION & LEARNING
# ============================================

# Every weekday at 8:30 PM - Validate signals from 3+ days ago (swing signals need time to resolve)
30 20 * * 1-5 cd $PROJECT_ROOT && $PYTHON_BIN validate_signals.py validate-all --min-age-days 3 >> scanner_output/logs/cron_validate.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_VALIDATE > /dev/null

# Every Sunday at 9:00 PM - Generate learning recommendations from accumulated outcomes
0 21 * * 0 cd $PROJECT_ROOT && $PYTHON_BIN validate_signals.py learn >> scanner_output/logs/cron_learn.log 2>&1 && curl -fsS -m 10 --retry 3 $HC_BASE/HC_UUID_VALIDATE > /dev/null
